cmake_minimum_required(VERSION 3.22.1)
project(pyrun_pygame_sdl LANGUAGES C CXX)

if(NOT ANDROID)
    message(FATAL_ERROR "PyRun's Pygame SDL stack must be cross-compiled with the Android NDK")
endif()

foreach(source_name IN ITEMS SDL SDL_IMAGE SDL_MIXER SDL_TTF)
    if(NOT DEFINED PYRUN_${source_name}_SOURCE_DIR OR
       NOT EXISTS "${PYRUN_${source_name}_SOURCE_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "Pinned ${source_name} source is missing")
    endif()
endforeach()

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS ON CACHE BOOL "Build the SDL bridge libraries as shared libraries" FORCE)
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -Wl,-z,max-page-size=16384")
set(CMAKE_MODULE_LINKER_FLAGS "${CMAKE_MODULE_LINKER_FLAGS} -Wl,-z,max-page-size=16384")

set(SDL_SHARED ON CACHE BOOL "Build SDL2 as a shared library" FORCE)
set(SDL_STATIC OFF CACHE BOOL "Do not duplicate SDL2 in static archives" FORCE)
set(SDL_TEST OFF CACHE BOOL "Do not build SDL2 test programs" FORCE)
set(SDL2_DISABLE_SDL2MAIN ON CACHE BOOL "Use PyRun's JNI entry point" FORCE)
set(SDL2_DISABLE_INSTALL ON CACHE BOOL "Do not install SDL2 outside the APK" FORCE)

set(SDL2IMAGE_VENDORED OFF CACHE BOOL "Use the bundled stb image backend" FORCE)
set(SDL2IMAGE_BACKEND_STB ON CACHE BOOL "Enable bundled PNG/JPEG decoding" FORCE)
set(SDL2IMAGE_SAMPLES OFF CACHE BOOL "Do not build SDL2_image samples" FORCE)
set(SDL2IMAGE_TESTS OFF CACHE BOOL "Do not build SDL2_image tests" FORCE)
set(SDL2IMAGE_AVIF OFF CACHE BOOL "Disable optional AVIF codec" FORCE)
set(SDL2IMAGE_JXL OFF CACHE BOOL "Disable optional JPEG-XL codec" FORCE)
set(SDL2IMAGE_TIF OFF CACHE BOOL "Disable optional TIFF codec" FORCE)
set(SDL2IMAGE_WEBP OFF CACHE BOOL "Disable optional WebP codec" FORCE)

set(SDL2MIXER_VENDORED ON CACHE BOOL "Use the bundled minimp3/stb Vorbis codecs" FORCE)
set(SDL2MIXER_SAMPLES OFF CACHE BOOL "Do not build SDL2_mixer samples" FORCE)
set(SDL2MIXER_CMD OFF CACHE BOOL "Do not build SDL2_mixer command tools" FORCE)
set(SDL2MIXER_FLAC OFF CACHE BOOL "Disable codecs not required by the first Pygame pack" FORCE)
set(SDL2MIXER_MOD OFF CACHE BOOL "Disable tracker formats requiring an extra source package" FORCE)
set(SDL2MIXER_GME OFF CACHE BOOL "Disable optional game-music-emu" FORCE)
set(SDL2MIXER_MIDI OFF CACHE BOOL "Disable optional MIDI backends" FORCE)
set(SDL2MIXER_OPUS OFF CACHE BOOL "Disable optional Opus decoder" FORCE)
set(SDL2MIXER_WAVPACK OFF CACHE BOOL "Disable optional WavPack decoder" FORCE)
set(SDL2MIXER_VORBIS STB CACHE STRING "Use the upstream bundled stb Vorbis decoder" FORCE)
set(SDL2MIXER_MP3 ON CACHE BOOL "Enable the upstream bundled minimp3 decoder" FORCE)
set(SDL2MIXER_MP3_MINIMP3 ON CACHE BOOL "Use minimp3 for MP3 decoding" FORCE)

set(SDL2TTF_VENDORED ON CACHE BOOL "Build the pinned FreeType source statically" FORCE)
set(SDL2TTF_HARFBUZZ OFF CACHE BOOL "Keep the first Android font stack minimal" FORCE)
set(SDL2TTF_SAMPLES OFF CACHE BOOL "Do not build SDL2_ttf samples" FORCE)
set(FT_DISABLE_ZLIB ON CACHE BOOL "Do not depend on an unbundled zlib package" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "Do not depend on an unbundled bzip2 package" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "Do not depend on an unbundled libpng package" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "Do not depend on an unbundled HarfBuzz package" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "Do not depend on an unbundled Brotli package" FORCE)

add_subdirectory("${PYRUN_SDL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/sdl2")
add_subdirectory("${PYRUN_SDL_IMAGE_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/sdl2_image")
add_subdirectory("${PYRUN_SDL_MIXER_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/sdl2_mixer")
add_subdirectory("${PYRUN_SDL_TTF_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/sdl2_ttf")
